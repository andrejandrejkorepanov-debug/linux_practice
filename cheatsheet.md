# Шпаргалка по Linux 
- ls -la(-ll): список файлов с правами
- cd: перейти в папку
- mkdir -p: создать папку, включая вложенные
- chmod 640: права владец rw, группа r, остальные ничего

## Git
git add файл - подготовить к сохранению
git commit -m "Что сделал" - сохранить версию
git push - отправить на GitHub
git stasus - что изменилось
git lig --oneline - история

## ssh-ключ для GitHub (делается один раз)
ssh-keygen -t ed25519 -C "почта" - создать пару ключей
cat ~/.ssh/id_ed25519.pub - показать открытый ключ (кладем на GitHub)

## Права и пользователи
sudo adduser имя - создать пользователя
sudo groupadd группа - создать группу
sudo usermod -aG группа имя - добавить пользователя в группу
groups имя - показать группы пользователя
su - имя - зайти под другим пользваотеля
exit - вернуться
sudo usermod -aG sudo имя - дать пользователю права адмнистратора
sudo deluser имя - удалить пользователя (файлы остаются)
sudo deluser --remove-home имя - удалить пользователя вместе с домашней папкой
sudo delgroup группа - удалить группу
id имя - информация о пользователе (его номер и группы)
## r=4, w=2, x=1 
sudo chggrp -R группа путь - сменить группу у папки и всего внутри
sudo chmod 750 папка - владелец всё, группа читает и заходит, остальные ничего
sudo chmod 640 файл - владелец читает и пишет, группа читает,, остальные ничего
echo "текст" | sudo tee файл - записать текст в файл с правами sudo
## Проверка соединения
ping -c 3 8.8.8.8 - есть ли интернет
ping -c 3 github.com - работает ли DNS 
sudo systemctl restart systemd-resolved - перезапустить DNS

## Процессы
ps aux - все процессы
htop - монитор (q  - выход)
sleep 600 & - запустить в фоне
pgrep имя - найти PID
kill PID - завершить, kill -9 PID - принудительно
Ctrl+C - прервать, Ctrl+Z - пауза
jobs, bg, fg - управление фоном
